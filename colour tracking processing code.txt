import processing.video.*;
import processing.serial.*;

Capture cam;
Serial myPort;
String lastState = "";

void setup() {
  size(640, 480);
  cam = new Capture(this, 640, 480);
  cam.start();

  // Serial setup (COM6, 9600 baud)
  myPort = new Serial(this, "COM6", 9600);
}

void draw() {
  if (cam.available()) {
    cam.read();
    image(cam, 0, 0);

    cam.loadPixels();
    int redCount = 0;
    int yellowCount = 0;

    for (int i = 0; i < cam.pixels.length; i++) {
      color c = cam.pixels[i];
      float r = red(c);
      float g = green(c);
      float b = blue(c);

      // crude red detection
      if (r > 150 && g < 100 && b < 100) {
        redCount++;
      }

      // crude yellow detection (red + green strong, blue weak)
      if (r > 150 && g > 150 && b < 100) {
        yellowCount++;
      }
    }

    fill(255);
    textSize(32);

    String currentState = lastState; // default to previous state

    if (yellowCount > 2000 && yellowCount > redCount) {
      currentState = "ON";
      fill(0, 255, 0);
      text("YELLOW = ON", 20, 40);
    } else if (redCount > 2000 && redCount > yellowCount) {
      currentState = "OFF";
      fill(255, 0, 0);
      text("RED = OFF", 20, 40);
    }

    // only send when state changes
    if (!currentState.equals(lastState)) {
      myPort.write(currentState + "\n");
      println("Sent " + currentState + " to Arduino");
      lastState = currentState;
    }
  }
}
